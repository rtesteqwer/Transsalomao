async function readKlebersomFleetState(sql: Awaited<ReturnType<typeof getSql>>): Promise<FleetState> {
  const driverRows = await sql<Record<string, unknown>>`
    select * from drivers
    where id = 'drv_d0d50a32b1'
       or lower(regexp_replace(trim(name), '[[:space:]]+', ' ', 'g')) = 'klebersom dutra da silva'
    order by case when id = 'drv_d0d50a32b1' then 0 else 1 end, name
    limit 1
  `;

  const driver = driverRows[0];
  if (!driver) {
    return { drivers: [], fleets: [], trips: [], reports: [], fuelings: [] };
  }

  const driverId = str(driver.id);
  const [trips, reports, directFuelings, allFuelings, allFleets] = await Promise.all([
    sql<Record<string, unknown>>`
      select * from trips
      where driver_id = ${driverId}
      order by date desc, code desc
    `,
    sql<Record<string, unknown>>`
      select * from reports
      where driver_id = ${driverId}
      order by created_at desc
    `,
    sql<Record<string, unknown>>`
      select * from fuelings
      where driver_id = ${driverId}
      order by date desc, created_at desc
    `,
    sql<Record<string, unknown>>`select * from fuelings order by date desc, created_at desc`,
    sql<Record<string, unknown>>`select * from fleets order by name`,
  ]);

  const fleetIds = new Set<string>();
  for (const row of trips) {
    const id = str(row.fleet_id);
    if (id) fleetIds.add(id);
  }
  for (const row of reports) {
    const id = str(row.fleet_id);
    if (id) fleetIds.add(id);
  }
  for (const row of directFuelings) {
    const id = str(row.fleet_id);
    if (id) fleetIds.add(id);
  }

  // Inclui também abastecimentos do(s) conjunto(s) do Klebersom que tenham sido
  // lançados sem driver_id, para espelhar tudo que pertence à operação dele.
  const fuelings = allFuelings.filter((row) => {
    const rowDriverId = str(row.driver_id);
    const rowFleetId = str(row.fleet_id);
    return rowDriverId === driverId || (!rowDriverId && fleetIds.has(rowFleetId));
  });
  for (const row of fuelings) {
    const id = str(row.fleet_id);
    if (id) fleetIds.add(id);
  }

  const fleets = allFleets.filter((row) => {
    const rowId = str(row.id);
    const rowName = str(row.name).trim().toLocaleLowerCase("pt-BR");
    return fleetIds.has(rowId) || rowName === 'volvo klebersom';
  });

  return {
    drivers: [mapDriver(driver)],
    fleets: fleets.map(mapFleet),
    trips: trips.map(mapTrip),
    reports: reports.map(mapReport),
    fuelings: fuelings.map(mapFueling),
  };
}

export const getKlebersomFleetState = createServerFn({ method: "GET" }).handler(
  async (): Promise<FleetState> => {
    const { assertPartnerOrAdminSession } = await import("@/lib/management-auth.server");
    const session = assertPartnerOrAdminSession();
    if (session.role !== "partner" && session.role !== "admin") {
      throw new Error("Usuário sem permissão para acessar os dados do Klebersom.");
    }
    const sql = await getSql();
    return readKlebersomFleetState(sql);
  },
);
